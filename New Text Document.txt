    <script>
        const socket = io();
        let currentUser = null;
        let currentUserId = null;
        let currentToken = null;
        let allPosts = [];
        let allUsers = [];
        let allMessages = [];
        let selectedChatUserId = null;
        let postImgData = null;
        let localStream = null;
        let peerConnection = null;
        let typingTimeout = null;
        let currentCommentPostId = null;
        let viewingUserId = null;

        function downloadImage(dataUrl, filename) {
            const link = document.createElement('a');
            link.href = dataUrl;
            link.download = filename || 'image.jpg';
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        function showNotification(message) {
            const notif = document.createElement('div');
            notif.className = 'notification-badge';
            notif.innerHTML = `<div class="flex items-center gap-2">${message}</div>`;
            document.body.appendChild(notif);
            setTimeout(() => notif.remove(), 3000);
        }

        function getLastMessage(userId) {
            const msgs = allMessages.filter(m => 
                (m.fromId === userId && m.toId === currentUserId) || 
                (m.fromId === currentUserId && m.toId === userId)
            );
            return msgs.length > 0 ? msgs[msgs.length - 1] : null;
        }

        function hasUnreadMessages(userId) {
            return allMessages.some(m => 
                m.fromId === userId && m.toId === currentUserId && m.read === 0
            );
        }

        // ============ SOCKET EVENTS ============
        socket.on('connect', () => console.log('✅ Connected'));

        socket.on('load-posts', (posts) => {
            allPosts = posts || [];
            renderFeed();
        });

        socket.on('new-post', (post) => {
            allPosts.unshift(post);
            renderFeed();
        });

        socket.on('post-deleted', (postId) => {
            allPosts = allPosts.filter(p => p.id !== postId);
            renderFeed();
        });

        socket.on('load-users', (users) => {
            allUsers = users || [];
            renderContacts();
            renderDiscover();
        });

        socket.on('users-updated', () => {
            socket.emit('auth', { token: currentToken });
        });

        socket.on('load-messages', (messages) => {
            allMessages = messages || [];
            if(selectedChatUserId) renderChat();
            updateContactsList();
            updateUnreadBadge();
        });

        socket.on('message-received', (msg) => {
            allMessages.push(msg);
            if(selectedChatUserId && (msg.from === selectedChatUserId || msg.fromId === selectedChatUserId)) {
                socket.emit('message-delivered', msg._id || msg.id);
                renderChat();
            }
            updateContactsList();
            updateUnreadBadge();
        });

        socket.on('notification', (notif) => {
            showNotification(`🔔 ${notif.message}`);
        });

        socket.on('user-typing', (data) => {
            if(data.fromId === selectedChatUserId) {
                document.getElementById('typing-status').innerHTML = `<div class="typing-indicator"><div class="typing-dot"></div><div class="typing-dot"></div><div class="typing-dot"></div></div> <span class="text-xs text-slate-500">typing...</span>`;
            }
        });

        socket.on('user-stop-typing', (data) => {
            document.getElementById('typing-status').innerHTML = '';
        });

        socket.on('incoming-call', (data) => {
            selectedChatUserId = data.fromUserId;
            document.getElementById('call-overlay').classList.remove('hidden');
            acceptCall();
        });

        socket.on('receive-offer', (data) => {
            handleReceiveOffer(data.offer);
        });

        socket.on('receive-answer', (data) => {
            if(peerConnection) {
                peerConnection.setRemoteDescription(new RTCSessionDescription(data.answer));
            }
        });

        socket.on('receive-ice-candidate', (data) => {
            if(peerConnection && data.candidate) {
                peerConnection.addIceCandidate(new RTCIceCandidate(data.candidate));

                alert('Error: ' + e.message);

        function updateContactsList() {
            const list = document.getElementById('contacts-list');
            
            if(allUsers.length === 0) {
                list.innerHTML = '<div class="text-center text-slate-400 py-4 text-sm">No users</div>';
                return;
            }

            // Sort by unread messages first, then by last message
            const sortedUsers = allUsers.filter(u => u.id !== currentUserId).sort((a, b) => {
                const aUnread = hasUnreadMessages(a.id);
                const bUnread = hasUnreadMessages(b.id);
                if (aUnread !== bUnread) return aUnread ? -1 : 1;
                
                const aMsg = getLastMessage(a.id);
                const bMsg = getLastMessage(b.id);
                const aTime = aMsg ? new Date(aMsg.createdAt) : new Date(0);
                const bTime = bMsg ? new Date(bMsg.createdAt) : new Date(0);
                return bTime - aTime;
            });

            list.innerHTML = sortedUsers.map(u => {
                const lastMsg = getLastMessage(u.id);
                const isUnread = hasUnreadMessages(u.id);
                const preview = lastMsg ? (lastMsg.text.length > 30 ? lastMsg.text.substring(0, 30) + '...' : lastMsg.text) : 'No messages';
                
                return `
                    <div onclick="openChat('${u.id}', '${u.username}')" class="p-3 rounded-lg hover:bg-slate-100 cursor-pointer transition ${isUnread ? 'bg-blue-50' : ''}">
                        <div class="flex items-center gap-3">
                            <div class="w-10 h-10 bg-sky-200 rounded-lg flex items-center justify-center font-bold text-sky-600 text-sm">${u.username.substring(0, 1)}</div>
                            <div class="flex-1 min-w-0">
                                <div class="flex justify-between items-center">
                                    <p class="font-bold text-sm truncate ${isUnread ? 'text-blue-600' : ''}">${u.username}</p>
                                    ${isUnread ? '<span class="w-2 h-2 bg-red-500 rounded-full"></span>' : ''}
                                </div>
                                <p class="text-xs text-slate-500 truncate">${preview}</p>
                            </div>
                        </div>
                    </div>
                `;
            }).join('');
        }

        async function createPeerConnection() {
            const config = {
                iceServers: [
                    { urls: ['stun:stun.l.google.com:19302'] },
                    { urls: ['stun:stun1.l.google.com:19302'] }
                ]
            };

            peerConnection = new RTCPeerConnection(config);

            if(localStream) {
                localStream.getTracks().forEach(track => {
                    peerConnection.addTrack(track, localStream);
                });
            }

            peerConnection.onicecandidate = (event) => {
                if(event.candidate) {
                    socket.emit('send-ice-candidate', {
                        toUserId: selectedChatUserId,
                        candidate: event.candidate
                    });
                }
            };

            peerConnection.ontrack = (event) => {
                document.getElementById('remote-video').srcObject = event.streams[0];
            };
        }

        async function acceptCall() {
            try {
                const constraints = { audio: true, video: true };
                localStream = await navigator.mediaDevices.getUserMedia(constraints);
                document.getElementById('local-video').srcObject = localStream;
                
                await createPeerConnection();
            } catch(e) {
                alert('Accept call error: ' + e.message);
            }
        }

        async function handleReceiveOffer(offer) {
            if(!peerConnection) {
                await createPeerConnection();
            }
            
            await peerConnection.setRemoteDescription(new RTCSessionDescription(offer));
            const answer = await peerConnection.createAnswer();
            await peerConnection.setLocalDescription(answer);
            socket.emit('send-answer', { fromUserId: selectedChatUserId, answer });
        }

        async function uploadAvatar() {
            const file = document.getElementById('avatar-upload').files[0];
            if(!file) return;

            const formData = new FormData();
            formData.append('image', file);

            try {
                const res = await fetch('/api/upload', {
                    method: 'POST',
                    body: formData
                });
                const data = await res.json();

                if(data.ok) {
                    await fetch('/api/update-profile', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                            userId: currentUserId,
                            avatar: data.url
                        })
                    });
                    alert('Avatar updated!');
                    location.reload();
                }
            } catch(err) {
                alert('Upload failed: ' + err.message);
            }
        }

        async function saveBio() {
            const bio = document.getElementById('profile-bio-edit').value.trim();

            try {
                await fetch('/api/update-profile', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        userId: currentUserId,
                        bio
                    })
                });
                alert('Bio saved!');
            } catch(err) {
                alert('Save failed');
            }
        }

        function renderProfile() {
            const currentUserData = allUsers.find(u => u.id === currentUserId);
            if(!currentUserData) return;

            document.getElementById('profile-name').textContent = currentUserData.username;
            document.getElementById('profile-bio-edit').value = currentUserData.bio || '';

            if(currentUserData.avatar) {
                document.getElementById('profile-avatar-display').innerHTML = `<img src="${currentUserData.avatar}" class="w-full h-full object-cover">`;
            }

            const userPosts = allPosts.filter(p => p.userId === currentUserId);
            document.getElementById('profile-posts').textContent = userPosts.length;
            document.getElementById('profile-followers').textContent = currentUserData.followers;

            const postsHtml = userPosts.map(p => `
                <div class="bg-white rounded-xl p-4 shadow-sm">
                    <p class="text-slate-700">${p.text}</p>
                    ${p.image ? `<div class="relative group"><img src="${p.image}" class="w-full h-40 object-cover rounded-lg mt-3"><button onclick="downloadImage('${p.image}', 'post-image.jpg')" class="absolute top-2 right-2 bg-slate-900 bg-opacity-70 text-white px-3 py-1 rounded-lg text-sm opacity-0 group-hover:opacity-100 transition">⬇️ Download</button></div>` : ''}
                </div>
            `).join('');

            document.getElementById('profile-posts-list').innerHTML = postsHtml || '<div class="text-center text-slate-400">No posts yet</div>';
        }

        // ============ DISCOVER ============
        function renderDiscover() {
            const grid = document.getElementById('discover-grid');
            const users = allUsers.filter(u => u.id !== currentUserId);

            if(users.length === 0) {
                grid.innerHTML = '<div class="text-center text-slate-400">No users</div>';
                return;
            }

            grid.innerHTML = users.map(u => `
                <div class="bg-white rounded-xl p-6 shadow-sm hover:shadow-md transition">
                    <div class="w-16 h-16 bg-sky-200 rounded-full mx-auto mb-3 flex items-center justify-center font-bold text-xl text-sky-600 cursor-pointer overflow-hidden" onclick="viewUserProfile('${u.id}')">
                        ${u.avatar ? `<img src="${u.avatar}" class="w-full h-full object-cover">` : u.username.substring(0, 1)}
                    </div>
                    <h3 class="font-bold text-center mb-1 cursor-pointer hover:text-sky-500" onclick="viewUserProfile('${u.id}')">${u.username}</h3>
                    <p class="text-xs text-slate-500 text-center mb-2">${u.followers} followers</p>
                    <p class="text-sm text-slate-600 text-center mb-4 line-clamp-2">${u.bio || 'No bio'}</p>
                    <button onclick="toggleAuraPlus2('${u.id}')" class="w-full ${u.isFollowing ? 'bg-slate-200 text-slate-600' : 'bg-sky-500 text-white'} py-2 rounded-lg font-bold hover:opacity-90 transition">
                        AURA+
                    </button>
                </div>
            `).join('');
        }
